rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // --- Funções Auxiliares ---
    function isSignedIn() {
      return request.auth != null;
    }

    function isSuperAdmin() {
      // UID do Super Admin (sem alteração)
      return request.auth.uid == "5Y7oZLEH4qebRr0HkCTTGaiZmcJ2";
    }

    function isOwnerOfSalon(salonId) {
      // Verifica se o usuário logado é o dono do salão (sem alteração)
      let salonData = get(/databases/$(database)/documents/salons/$(salonId)).data;
      return isSignedIn() && request.auth.uid == salonData.ownerId;
    }
    
    // (NOVA FUNÇÃO) Retorna o UID do usuário associado a um professionalId
    // Isso é usado para verificar se o profissional logado é o mesmo do agendamento.
    function getUserIdForProfessional(professionalId) {
        return get(/databases/$(database)/documents/professionals/$(professionalId)).data.userId;
    }

    // --- Regras por Coleção ---

    // REGRA PARA 'salons': CORRIGIDA
    match /salons/{salonId} {
      // Permite que o app CONSULTE a coleção para encontrar o salão do usuário no login.
      allow list: if isSignedIn();
      // Apenas o dono ou super admin pode LER ou ESCREVER em um documento de salão específico.
      allow get, write: if isSuperAdmin() || isOwnerOfSalon(salonId);
    }

    // REGRA PARA 'professionals':
    match /professionals/{profId} {
      // Permite que qualquer usuário logado CONSULTE a coleção para encontrar seu perfil no login.
      allow list: if isSignedIn(); 
      // Permite que dono, super admin ou o próprio profissional LEIA o documento.
      allow get: if isSuperAdmin() || isOwnerOfSalon(resource.data.salonId) || request.auth.uid == resource.data.userId;
      // Apenas dono e super admin podem criar, editar ou deletar profissionais.
      allow write: if isSuperAdmin() || isOwnerOfSalon(request.resource.data.salonId);
    }

    // REGRAS PARA 'clients' & 'services':
    // Profissionais precisam ler essas informações para agendar. Apenas donos podem modificar.
    match /clients/{clientId} {
      allow read: if isSignedIn(); // A query no app já filtra por salonId, tornando a leitura segura.
      allow write: if isSuperAdmin() || isOwnerOfSalon(request.resource.data.salonId);
    }
    
    match /services/{serviceId} {
      allow read: if isSignedIn(); // A query no app já filtra por salonId.
      allow write: if isSuperAdmin() || isOwnerOfSalon(request.resource.data.salonId);
    }
    
    // REGRA PARA 'appointments':
    match /appointments/{appointmentId} {
      // Leitura é permitida para usuários logados; o app filtra o que é relevante.
      allow read: if isSignedIn();
      
      // Permite CRIAR um agendamento se for o dono ou o próprio profissional.
      allow create: if isSuperAdmin() || isOwnerOfSalon(request.resource.data.salonId) 
                     || request.auth.uid == getUserIdForProfessional(request.resource.data.professionalId);
      
      // Permite EDITAR/DELETAR se for o dono ou o profissional que JÁ ESTÁ no agendamento.
      allow update, delete: if isSuperAdmin() || isOwnerOfSalon(resource.data.salonId)
                             || request.auth.uid == getUserIdForProfessional(resource.data.professionalId);
    }
  }
}
