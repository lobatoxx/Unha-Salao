rules_version = '2';

service cloud.firestore {
  match databases{database}documents {

     --- Funções Auxiliares ---
    function isSuperAdmin() {
       UID do Super Admin
      return request.auth.uid == 5Y7oZLEH4qebRr0HkCTTGaiZmcJ2;
    }

    function isSignedIn() {
      return request.auth != null;
    }

    function getUserContext() {
      return get(databases$(database)documentsusers$(request.auth.uid)).data;
    }

    function isOwnerOfSalon(salonId) {
      return request.auth.uid == get(databases$(database)documentssalons$(salonId)).data.ownerId;
    }

     --- Regras por Coleção ---

    match users{userId} {
       Super Admin ou o próprio utilizador podem lerescrever.
      allow read, write if isSuperAdmin()  (isSignedIn() && request.auth.uid == userId);
    }

    match salons{salonId} {
       Super Admin pode ler a lista toda. Utilizadores normais também para a migração.
      allow list if isSuperAdmin()  isSignedIn(); 
       Super Admin ou um membro do salão podem ler os detalhes.
      allow get if isSuperAdmin()  (isSignedIn() && getUserContext().salonId == salonId);
       Super Admin ou o dono podem escrever.
      allow write if isSuperAdmin()  (isSignedIn() && isOwnerOfSalon(salonId));
    }

     Coleções filhas de salões
    match clients{clientId} {
       Super Admin ou um membro do salão podem ler.
      allow read if isSuperAdmin()  (isSignedIn() && getUserContext().salonId == resource.data.salonId);
       Super Admin ou o dono podem escrever.
      allow write if isSuperAdmin()  (isSignedIn() && isOwnerOfSalon(request.resource.data.salonId));
    }

    match services{serviceId} {
      allow read if isSuperAdmin()  (isSignedIn() && getUserContext().salonId == resource.data.salonId);
      allow write if isSuperAdmin()  (isSignedIn() && isOwnerOfSalon(request.resource.data.salonId));
    }
    
    match professionals{profId} {
      allow read if isSuperAdmin()  (isSignedIn() && getUserContext().salonId == resource.data.salonId);
      allow write if isSuperAdmin()  (isSignedIn() && isOwnerOfSalon(request.resource.data.salonId));
    }

    match appointments{appointmentId} {
      allow read if isSuperAdmin()  (isSignedIn() && getUserContext().salonId == resource.data.salonId);
      allow write if isSuperAdmin()  (isSignedIn() &&
                     (
                       isOwnerOfSalon(request.resource.data.salonId) 
                       (getUserContext().role == 'professional' && getUserContext().salonId == request.resource.data.salonId)
                     ));
    }
  }
}
